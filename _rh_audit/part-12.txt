function extractJsonBlock(text) {
  if (!text) return null;
  const startIdx = text.indexOf("{");
  if (startIdx === -1) return null;
  let depth = 0;

  let escape2 = false;
  for (let i = startIdx; i < text.length; i++) {
    const char = text[i];
    if (escape2) {
      escape2 = false;
      continue;
    }
    if (char === "\\") {
      escape2 = true;
      continue;
    }
    if (char === '"') {
      inString = !inString;
      continue;
    }
    if (!inString) {
      if (char === "{") depth++;
      else if (char === "}") {
        depth--;
        if (depth === 0) {
          return {
            jsonStr: text.substring(startIdx, i + 1),
            start: startIdx,
            end: i + 1
          };
        }
      }
    }
  }
  return null;
}
