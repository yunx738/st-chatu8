function eidtJSON(obj) {
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      if (typeof obj[key] === "object" && obj[key] !== null) {
        eidtJSON(obj[key]);
      } else {
        if (key.includes("seed")) {
          obj[key] = "%seed%";
        }
        if (key == "steps") {
          obj[key] = "%steps%";
        }
        if (key == "cfg") {
          obj[key] = "%cfg_scale%";
        }
        if (key == "sampler_name") {
          obj[key] = "%sampler_name%";
        }
        if (key == "width") {
          obj[key] = "%width%";
        }
        if (key == "height") {
          obj[key] = "%height%";
        }
        if (key == "ckpt_name") {
          obj[key] = "%MODEL_NAME%";
        }
        if (key == "positive") {
          obj[key] = "%prompt%";
        }
        if (key == "negative") {
          obj[key] = "%negative_prompt%";
        }
        if (key == "text" && obj[key] == "\u6B63\u9762") {
          obj[key] = "%prompt%";
        }
        if (key == "text" && obj[key] == "\u8D1F\u9762") {
          obj[key] = "%negative_prompt%";
        }
      }
    }
  }
  return obj;
}
